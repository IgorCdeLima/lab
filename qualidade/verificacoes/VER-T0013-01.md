Script usado (scratchpad, fora do Git; essencial para reexecutar): cliente `http.client` do Python que monta multipart a mao e imprime status e marcadores do HTML.

```python
import http.client, io, uuid
def req(m, p, corpo=None, cab=None):
    c = http.client.HTTPConnection("127.0.0.1", 8113, timeout=30); c.request(m, p, body=corpo, headers=cab or {})
    r = c.getresponse(); return r.status, dict((k.lower(), v) for k, v in r.getheaders()), r.read()
def multipart(campos, arquivo=None):
    b = uuid.uuid4().hex; out = io.BytesIO()
    for k, v in campos.items():
        out.write(f'--{b}\r\nContent-Disposition: form-data; name="{k}"\r\n\r\n'.encode())
        out.write(v.encode("utf-8", "surrogatepass")); out.write(b"\r\n")
    if arquivo:
        n, dados, tipo = arquivo
        out.write(f'--{b}\r\nContent-Disposition: form-data; name="imagem"; filename="{n}"\r\nContent-Type: {tipo}\r\n\r\n'.encode() + dados + b"\r\n")
    out.write(f"--{b}--\r\n".encode()); return out.getvalue(), {"Content-Type": f"multipart/form-data; boundary={b}"}
# para cada caso: s, h, d = req("POST", "/produtos", *multipart(campos)); conferir s != 500, role="alert", autofocus, escape
```
